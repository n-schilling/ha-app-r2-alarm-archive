abi <abi/3.0>,
#include <tunables/global>

# Talks to Home Assistant, Cloudflare and the MQTT broker; writes its queue,
# state and cache to /data and quarantined objects to its folder in
# /addon_configs. s6-overlay runs as root and hands both to user r2archive,
# as which the archive runs. Measured on Home Assistant OS in complain mode
# before it enforced.
profile r2_alarm_archive flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # s6-overlay, and the hand-over of the folders to the archive's user
  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,

  network inet dgram,
  network inet6 dgram,
  network inet stream,
  network inet6 stream,
  network unix,

  # s6 stops its services; the Supervisor stops the container
  signal (send, receive),

  # s6-overlay
  /init ix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /run/** rwk,
  /run/ r,

  # Programs and libraries of the image
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/local/bin/** rix,
  /usr/lib/** rm,
  /usr/local/lib/** rm,
  /lib/** rm,
  /usr/share/** r,
  /etc/** r,

  # bashio
  /tmp/ r,
  /tmp/** rwk,

  # Options, queue, state, cache; the quarantine
  /data/ r,
  /data/** rwk,
  /config/ rw,
  /config/** rwk,

  @{PROC}/@{pid}/** r,
  @{PROC}/ r,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,
  /dev/tty rw,

  deny /proc/sys/** w,
  deny /sys/** w,
}
